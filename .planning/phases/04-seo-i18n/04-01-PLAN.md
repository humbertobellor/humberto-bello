2. **Add data-i18n replacement function:**
```javascript
function replaceI18n(html, data, prefix = '') {
  return html.replace(/data-i18n="([^"]+)"/g, (_, key) => {
    // Remove the attribute and set textContent
    const parts = key.split('.');
    let value = data;
    for (const part of parts) {
      // Handle array notation: "items[0]" -> "items", 0
      const arrMatch = part.match(/^(\w+)\[(\d+)\]$/);
      if (arrMatch) {
        value = value[arrMatch[1]];
        if (value) value = value[parseInt(arrMatch[2])];
      } else {
        value = value ? value[part] : undefined;
      }
    }
    if (typeof value === 'string') {
      return `>${value}<`;
    }
    return `>${key}<`; // fallback: keep key as text
  });
}
```
